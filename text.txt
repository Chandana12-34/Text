import pandas as pd
import json
import os
import time
import re

from tqdm import tqdm
from dotenv import load_dotenv
from openai import OpenAI

# =========================================================
# LOAD ENV VARIABLES
# =========================================================

load_dotenv()

API_KEY = os.getenv("API_KEY")

# =========================================================
# CREATE OPENAI CLIENT
# =========================================================

client = OpenAI(
    api_key=API_KEY,
    base_url="https://api.llm.cib.echonet/v1/openai"
)

# =========================================================
# FILE PATHS
# =========================================================

INPUT_FILE = "input/ritm_input.xlsx"
OUTPUT_FILE = "output/ritm_output.xlsx"

# =========================================================
# READ EXCEL FILE
# =========================================================

df = pd.read_excel(INPUT_FILE)

# =========================================================
# AUTO COLUMN DETECTION
# =========================================================

def find_column(keyword_list):

    for col in df.columns:

        lower = col.lower()

        for keyword in keyword_list:

            if keyword in lower:
                return col

    return None

NUMBER_COL = find_column(["number"])
SHORT_DESC_COL = find_column(["short"])
DESCRIPTION_COL = find_column(["description"])
COMMENTS_COL = find_column(["comments", "work notes"])
USER_TYPE_COL = find_column(["new vs old"])

print("Detected Columns:")
print(NUMBER_COL)
print(SHORT_DESC_COL)
print(DESCRIPTION_COL)
print(COMMENTS_COL)
print(USER_TYPE_COL)

# =========================================================
# CLEAN COMMENTS FUNCTION
# =========================================================

def clean_comments(text):

    if pd.isna(text):
        return ""

    text = str(text)

    patterns = [
        r"Infos:.*",
        r"State - Open.*",
        r"State - On Hold.*",
        r"closed after 72 hrs.*",
        r"APS L1 Support.*"
    ]

    for pattern in patterns:
        text = re.sub(pattern, "", text)

    return text.strip()

# =========================================================
# SYSTEM PROMPT
# =========================================================

SYSTEM_PROMPT = """
You are an expert IT Service Management analyst.

Analyze ServiceNow RITM tickets.

Generate:

1. Issue Summary
2. Solution Provided
3. Business-Relevant Category
4. System Component
5. User Type
6. Actionable Insights

RULES:

- Use concise business-friendly language.
- Ignore system-generated metadata.
- Read comments/work notes carefully.
- Extract actual analyst resolution.
- Mention if issue closed after inactivity.

Allowed Categories:
- User Knowledge Gap
- Access Issue
- Data Confusion
- System Bug
- Process Clarification
- Configuration Error
- Release/Change Impact
- Other

Actionable Insights examples:
- Training Opportunities
- Process Improvements
- Application Improvements
- Automation Opportunities

Return ONLY VALID JSON.

JSON FORMAT:

{
    "Issue Summary": "",
    "Solution Provided": "",
    "Business-Relevant Category": "",
    "System Component": "",
    "User Type": "",
    "Actionable Insights": ""
}
"""

# =========================================================
# ANALYZE SINGLE TICKET
# =========================================================

def analyze_ticket(row):

    ticket_number = str(row.get(NUMBER_COL, ""))
    short_desc = str(row.get(SHORT_DESC_COL, ""))
    description = str(row.get(DESCRIPTION_COL, ""))
    comments = clean_comments(row.get(COMMENTS_COL, ""))
    user_type = str(row.get(USER_TYPE_COL, ""))

    user_prompt = f"""
    Ticket Number:
    {ticket_number}

    Short Description:
    {short_desc}

    Description:
    {description}

    Comments and Work Notes:
    {comments}

    User Type:
    {user_type}
    """

    try:

        response = client.chat.completions.create(
            model="mistral-medium-2508-ITG",
            temperature=0.2,
            messages=[
                {
                    "role": "system",
                    "content": SYSTEM_PROMPT
                },
                {
                    "role": "user",
                    "content": user_prompt
                }
            ]
        )

        content = response.choices[0].message.content

        content = (
            content
            .replace("```json", "")
            .replace("```", "")
            .strip()
        )

        parsed = json.loads(content)

        return {
            "A - RITM":
                f"RITM{ticket_number}"
                if "RITM" not in ticket_number
                else ticket_number,

            "B - Issue Summary":
                parsed.get("Issue Summary", ""),

            "C - Solution Provided":
                parsed.get("Solution Provided", ""),

            "D - Category":
                parsed.get("Business-Relevant Category", ""),

            "E - System Component":
                parsed.get("System Component", ""),

            "F - User Type":
                parsed.get("User Type", ""),

            "G - Actionable Insights":
                parsed.get("Actionable Insights", "")
        }

    except Exception as e:

        print(f"Error Processing Ticket {ticket_number}")
        print(e)

        return {
            "A - RITM": ticket_number,
            "B - Issue Summary": "ERROR",
            "C - Solution Provided": "ERROR",
            "D - Category": "ERROR",
            "E - System Component": "ERROR",
            "F - User Type": "ERROR",
            "G - Actionable Insights": "Manual Review Needed"
        }

# =========================================================
# PROCESS ALL TICKETS
# =========================================================

results = []

for _, row in tqdm(df.iterrows(), total=len(df)):

    result = analyze_ticket(row)

    results.append(result)

    time.sleep(1)

# =========================================================
# CREATE OUTPUT DATAFRAME
# =========================================================

output_df = pd.DataFrame(results)

# =========================================================
# EXPORT EXCEL
# =========================================================

with pd.ExcelWriter(
    OUTPUT_FILE,
    engine="xlsxwriter"
) as writer:

    output_df.to_excel(
        writer,
        sheet_name="RITM Analysis",
        index=False
    )

    workbook = writer.book
    worksheet = writer.sheets["RITM Analysis"]

    # HEADER FORMAT
    header_format = workbook.add_format({
        "bold": True,
        "bg_color": "#CFE2F3",
        "border": 1,
        "text_wrap": True,
        "valign": "top"
    })

    # CELL FORMAT
    cell_format = workbook.add_format({
        "text_wrap": True,
        "valign": "top",
        "border": 1
    })

    # FORMAT HEADERS
    for col_num, value in enumerate(output_df.columns.values):
        worksheet.write(0, col_num, value, header_format)

    # COLUMN WIDTHS
    worksheet.set_column("A:A", 20, cell_format)
    worksheet.set_column("B:B", 40, cell_format)
    worksheet.set_column("C:C", 80, cell_format)
    worksheet.set_column("D:D", 25, cell_format)
    worksheet.set_column("E:E", 35, cell_format)
    worksheet.set_column("F:F", 15, cell_format)
    worksheet.set_column("G:G", 70, cell_format)

print("\n===================================")
print("RITM ANALYSIS COMPLETED")
print(f"Output saved to: {OUTPUT_FILE}")
print("===================================")
